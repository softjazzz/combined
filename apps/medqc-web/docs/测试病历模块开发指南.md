## 🛠️ 开发环境搭建

### 前置条件
- Node.js >= 18.0.0
- Python >= 3.10
- PostgreSQL >= 13
- Redis >= 6.0
- Docker & Docker Compose (可选)

### 快速启动
```bash
# 1. 克隆项目
git clone <repository-url>
cd MedQC0817

# 2. 安装前端依赖
npm install
# 或
pnpm install

# 3. 安装后端依赖
cd production_system
pip install -r requirements.txt

# 4. 配置环境变量
cp .env.example .env
```

### 3. 测试执行引擎

#### 3.1 测试执行服务
```python
# services/test_execution_service.py
from typing import List, Dict, Any
import asyncio
from datetime import datetime

class TestExecutionService:
    def __init__(self, qc_engine, rule_engine):
        self.qc_engine = qc_engine
        self.rule_engine = rule_engine
    
    async def execute_scenario(self, scenario_id: str, parameters: Dict[str, Any] = None) -> str:
        """执行测试场景"""
        # 1. 获取场景配置
        scenario = await self.get_scenario(scenario_id)
        
        # 2. 创建执行记录
        execution = await self.create_execution_record(scenario_id)
        
        # 3. 异步执行测试
        asyncio.create_task(self._execute_scenario_async(execution.id, scenario, parameters))
        
        return execution.id
    
    async def _execute_scenario_async(self, execution_id: str, scenario: TestScenario, parameters: Dict[str, Any]):
        """异步执行测试场景"""
        try:
            # 更新执行状态
            await self.update_execution_status(execution_id, 'running')
            
            results = []
            
            # 按顺序执行每个测试病历
            for record_config in scenario.record_configs:
                record_result = await self._execute_record_test(record_config, scenario.rule_configs)
                results.append(record_result)
            
            # 分析整体结果
            overall_result = self._analyze_overall_result(results)
            
            # 保存执行结果
            await self.save_execution_result(execution_id, {
                'status': 'completed',
                'results': results,
                'overall_result': overall_result,
                'end_time': datetime.utcnow()
            })
            
        except Exception as e:
            # 处理执行异常
            await self.save_execution_result(execution_id, {
                'status': 'failed',
                'error': str(e),
                'end_time': datetime.utcnow()
            })
    
    async def _execute_record_test(self, record_config: Dict, rule_configs: List[Dict]) -> Dict:
        """执行单个病历测试"""
        record = await self.get_test_record(record_config['record_id'])
        
        # 应用质控规则
        qc_result = await self.qc_engine.analyze(
            record_id=record.id,
            content=record.content,
            rules=rule_configs
        )
        
        # 对比预期结果
        comparison = self._compare_with_expected(qc_result, record.expected_results)
        
        return {
            'record_id': record.id,
            'record_name': record.name,
            'qc_result': qc_result,
            'expected_result': record.expected_results,
            'comparison': comparison,
            'passed': comparison['overall_match']
        }
```

#### 3.2 结果分析器
```python
# utils/result_analyzer.py
class ResultAnalyzer:
    @staticmethod
    def analyze_test_results(execution_results: List[Dict]) -> Dict[str, Any]:
        """分析测试结果"""
        total_tests = len(execution_results)
        passed_tests = sum(1 for result in execution_results if result['passed'])
        failed_tests = total_tests - passed_tests
        
        # 计算通过率
        pass_rate = (passed_tests / total_tests) * 100 if total_tests > 0 else 0
        
        # 分析失败原因
        failure_reasons = []
        for result in execution_results:
            if not result['passed']:
                failure_reasons.extend(result['comparison']['differences'])
        
        # 统计失败原因
        failure_stats = {}
        for reason in failure_reasons:
            failure_stats[reason] = failure_stats.get(reason, 0) + 1
        
        # 性能分析
        performance_metrics = {
            'avg_execution_time': sum(r.get('execution_time', 0) for r in execution_results) / total_tests,
            'max_execution_time': max(r.get('execution_time', 0) for r in execution_results),
            'min_execution_time': min(r.get('execution_time', 0) for r in execution_results)
        }
        
        return {
            'summary': {
                'total_tests': total_tests,
                'passed_tests': passed_tests,
                'failed_tests': failed_tests,
                'pass_rate': pass_rate
            },
            'failure_analysis': {
                'failure_stats': failure_stats,
                'common_issues': sorted(failure_stats.items(), key=lambda x: x[1], reverse=True)[:5]
            },
            'performance_metrics': performance_metrics,
            'recommendations': ResultAnalyzer._generate_recommendations(pass_rate, failure_stats)
        }
    
    @staticmethod
    def _generate_recommendations(pass_rate: float, failure_stats: Dict) -> List[str]:
        """生成改进建议"""
        recommendations = []
        
        if pass_rate < 80:
            recommendations.append("测试通过率较低，建议检查质控规则的准确性")
        
        if failure_stats:
            most_common_issue = max(failure_stats.items(), key=lambda x: x[1])
            recommendations.append(f"最常见的问题是：{most_common_issue[0]}，建议重点关注")
        
        return recommendations
```

```python
class TestRecordService:
    """测试病历服务类

    负责处理测试病历的业务逻辑，包括创建、更新、删除和查询操作。
    """

    def __init__(self, repository: TestRecordRepository):
        self.repository = repository

    async def create_record(self, record_data: CreateRecordRequest) -> TestMedicalRecord:
        """创建测试病历

        Args:
            record_data: 病历创建请求数据

        Returns:
            创建的测试病历对象

        Raises:
            ValidationError: 当数据验证失败时
            DuplicateError: 当病历名称重复时
        """
        # 验证数据
        self._validate_record_data(record_data)

        # 检查重复
        existing = await self.repository.get_by_name(record_data.name)
        if existing:
            raise DuplicateError(f"Record with name '{record_data.name}' already exists")

        # 创建记录
        record = TestMedicalRecord(**record_data.dict())
        return await self.repository.create(record)

    def _validate_record_data(self, record_data: CreateRecordRequest) -> None:
        """验证病历数据"""
        if not record_data.name.strip():
            raise ValidationError("Record name cannot be empty")

        if len(record_data.content) < 100:
            raise ValidationError("Record content is too short")
```

```typescript
  const record = ref<TestMedicalRecord | null>(recordCache.get(id) || null)
```

```typescript
export function handleApiError(error: any): never {
  if (error.response) {
    const { status, data } = error.response
    throw new ApiError(status, data.message || '请求失败', data.code)
  } else if (error.request) {
    throw new ApiError(0, '网络连接失败')
  } else {
    throw new ApiError(0, error.message || '未知错误')
  }
}
```
