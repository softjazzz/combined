## 1. 开发环境准备

### 1.1 技术栈

- **Vue 3.4+** - 使用 Composition API 和 `<script setup>` 语法
- **TypeScript 5.x** - 强类型支持
- **Pinia 2.x** - 状态管理
- **Ant Design Vue 4.2.6** - UI 组件库
- **Vite 6.x** - 构建工具

## 3. 核心组件开发指南

### 3.1 DemoControlPanel.vue（演示控制面板）

#### 功能需求
- 提供演示的全局控制（开始、暂停、停止、重播）
- 步骤导航（上一步、下一步、跳转）
- 进度显示（圆形进度条）
- 快捷操作菜单（导入、导出、配置、清除）

#### 完整代码示例

```vue
<template>
  <div class="demo-control-panel">
    <a-card :bordered="false" size="small" class="control-card">
      <a-row :gutter="16" align="middle">
        <!-- 左侧：演示控制按钮 -->
        <a-col :flex="1">
          <a-space :size="8" wrap>
            <!-- 主要控制按钮组 -->
            <a-button-group>
              <a-button
                type="primary"
                :icon="h(PlayCircleOutlined)"
                :disabled="!demoStore.canStartDemo || demoStore.isRunning"
                :loading="demoStore.isRunning && !demoStore.isPaused"
                @click="handleStart"
              >
                开始演示
              </a-button>
              <a-button
                :icon="h(demoStore.isPaused ? CaretRightOutlined : PauseOutlined)"
                :disabled="!demoStore.isRunning"
                @click="handleTogglePause"
              >
                {{ demoStore.isPaused ? '继续' : '暂停' }}
              </a-button>
              <a-button
                danger
                :icon="h(StopOutlined)"
                :disabled="!demoStore.isRunning"
                @click="handleStop"
              >
                停止
              </a-button>
              <a-button
                :icon="h(ReloadOutlined)"
                @click="handleReplay"
              >
                重播
              </a-button>
            </a-button-group>

            <a-divider type="vertical" />

            <!-- 步骤导航 -->
            <a-button
              size="small"
              :icon="h(LeftOutlined)"
              :disabled="currentStepIndex === 0"
              @click="handlePrevStep"
            >
              上一步
            </a-button>

            <a-select
              v-model:value="demoStore.currentStep"
              style="width: 150px"
              size="small"
            >
              <a-select-option value="target">
                <FileSearchOutlined />
                测试对象选择
              </a-select-option>
              <a-select-option value="rules">
                <SettingOutlined />
                规则配置
              </a-select-option>
              <a-select-option value="run">
                <ThunderboltOutlined />
                运行与结果
              </a-select-option>
            </a-select>

            <a-button
              size="small"
              :icon="h(RightOutlined)"
              :disabled="currentStepIndex === 2"
              @click="handleNextStep"
            >
              下一步
            </a-button>
          </a-space>
        </a-col>

        <!-- 右侧：进度显示 + 快捷操作 -->
        <a-col :flex="0">
          <a-space :size="16">
            <!-- 进度显示 -->
            <a-tooltip title="当前进度">
              <a-progress
                type="circle"
                :percent="demoStore.progress"
                :width="40"
                :stroke-width="8"
                :status="demoStore.error ? 'exception' : 'normal'"
              />
            </a-tooltip>

            <!-- 快捷操作菜单 -->
            <a-dropdown>
              <a-button :icon="h(MoreOutlined)">
                更多
              </a-button>
              <template #overlay>
                <a-menu @click="handleMenuClick">
                  <a-menu-item key="import">
                    <ImportOutlined />
                    导入演示状态
                  </a-menu-item>
                  <a-menu-item key="export">
                    <ExportOutlined />
                    导出演示状态
                  </a-menu-item>
                  <a-menu-divider />
                  <a-menu-item key="config">
                    <SettingOutlined />
                    演示配置
                  </a-menu-item>
                  <a-menu-item key="clear" danger>
                    <DeleteOutlined />
                    清除状态
                  </a-menu-item>
                </a-menu>
              </template>
            </a-dropdown>
          </a-space>
        </a-col>
      </a-row>
    </a-card>
  </div>
</template>

<script setup lang="ts">
import { h, computed } from 'vue'
import { message, Modal } from 'ant-design-vue'
import { useQCDemoStore } from '@/stores/qcDemo'
import {
  PlayCircleOutlined,
  PauseOutlined,
  CaretRightOutlined,
  StopOutlined,
  ReloadOutlined,
  LeftOutlined,
  RightOutlined,
  MoreOutlined,
  ImportOutlined,
  ExportOutlined,
  SettingOutlined,
  DeleteOutlined,
  FileSearchOutlined,
  ThunderboltOutlined
} from '@ant-design/icons-vue'

const demoStore = useQCDemoStore()

// 计算当前步骤索引
const currentStepIndex = computed(() => {
  const steps = ['target', 'rules', 'run']
  return steps.indexOf(demoStore.currentStep)
})

// 事件处理
const handleStart = () => {
  if (!demoStore.canStartDemo) {
    message.warning('请先选择测试对象和质控规则')
    return
  }
  demoStore.startDemo()
}

const handleTogglePause = () => {
  demoStore.togglePause()
  message.info(demoStore.isPaused ? '演示已暂停' : '演示已继续')
}

const handleStop = () => {
  Modal.confirm({
    title: '确认停止演示？',
    content: '停止后将清除当前演示进度',
    okText: '确认',
    cancelText: '取消',
    onOk: () => {
      demoStore.stopDemo()
      message.success('演示已停止')
    }
  })
}

const handleReplay = () => {
  Modal.confirm({
    title: '确认重播演示？',
    content: '重播将重置所有演示状态',
    okText: '确认',
    cancelText: '取消',
    onOk: () => {
      demoStore.replayDemo()
      message.success('演示已重置，可以重新开始')
    }
  })
}

const handlePrevStep = () => {
  demoStore.previousStep()
}

const handleNextStep = () => {
  demoStore.nextStep()
}

const handleMenuClick = ({ key }: { key: string }) => {
  switch (key) {
    case 'import':
      demoStore.importDemoState()
      break
    case 'export':
      demoStore.exportDemoState()
      break
    case 'config':
      demoStore.showConfigModal = true
      break
    case 'clear':
      Modal.confirm({
        title: '确认清除所有状态？',
        content: '此操作将清除所有演示数据，包括通知和申诉记录',
        okText: '确认',
        cancelText: '取消',
        okType: 'danger',
        onOk: () => {
          demoStore.clearDemoState()
          message.success('状态已清除')
        }
      })
      break
  }
}
</script>

<style scoped lang="less">
.demo-control-panel {
  margin-bottom: 16px;

  .control-card {
    box-shadow: 0 2px 8px rgba(0, 0, 0, 0.08);
  }
}
</style>
```

#### 关键点说明

1. **按钮状态控制**：
   - `开始演示`：仅在 `canStartDemo` 为 true 且未运行时可用
   - `暂停/继续`：仅在运行时可用
   - `停止`：仅在运行时可用
   - `重播`：始终可用

2. **步骤导航**：
   - 使用 `currentStepIndex` 计算当前步骤索引
   - `上一步` 在第一步时禁用
   - `下一步` 在最后一步时禁用

3. **确认对话框**：
   - 停止、重播、清除状态等危险操作需要确认

- 展示通知列表（使用 `a-list`）

- 单项操作（查看、确认、申诉、忽略、删除）

```vue
<template>

    <!-- 顶部操作栏 -->
    <div class="manager-header">

            <a-select-option value="">全部</a-select-option>

          <a-select

          <a-button size="small" :icon="h(ReloadOutlined)" @click="handleRefresh">
            刷新
          </a-button>
        </a-space>

        <!-- 批量操作提示 -->
        <a-alert
          v-if="selectedIds.length > 0"
          type="info"
          :message="`已选择 ${selectedIds.length} 项`"
          closable
          @close="selectedIds = []"
        >
          <template #action>
            <a-space :size="8">
              <a-button size="small" @click="handleBatchMarkRead">
                标记已读
              </a-button>
              <a-button size="small" @click="handleBatchIgnore">
                批量忽略
              </a-button>
              <a-button size="small" danger @click="handleBatchDelete">
                批量删除
              </a-button>
            </a-space>
          </template>
        </a-alert>
      </a-space>
    </div>

    <!-- 通知列表 -->
    <a-list
      :data-source="filteredNotifications"
      :pagination="pagination"
      size="small"
      class="notification-list"
    >
      <template #header>
        <div class="list-header">
          <a-statistic
            title="未读通知"
            :value="demoStore.unreadNotificationsCount"
            :value-style="{ color: '#1890ff', fontSize: '20px' }"
          >
            <template #prefix>
              <BellOutlined />
            </template>
          </a-statistic>
        </div>
      </template>

      <template #renderItem="{ item }">
        <a-list-item
          :class="['notification-item', { 'unread': !item.isRead }]"
          @click="handleView(item)"
        >
          <template #actions>
            <a-dropdown>
              <a-button type="text" size="small" :icon="h(MoreOutlined)" />
              <template #overlay>
                <a-menu @click="({ key }) => handleAction(key, item)">
                  <a-menu-item key="view">
                    <EyeOutlined />
                    查看详情
                  </a-menu-item>
                  <a-menu-item key="confirm">
                    <CheckOutlined />

                  <a-menu-item key="appeal">
                    <FileTextOutlined />
                    申诉
                  </a-menu-item>

                    忽略
                  </a-menu-item>
                  <a-menu-item key="delete" danger>
                    <DeleteOutlined />
                    删除
                  </a-menu-item>

          <a-list-item-meta>

                @change="(e) => handleSelectChange(e, item.id)"

              />
              <a-badge :dot="!item.isRead" :offset="[-5, 5]">

                  <template #icon>
                    <component :is="getTypeIcon(item.type)" />

              </a-badge>
            </template>

            <template #title>
              <a-space>
                <span :class="{ 'font-bold': !item.isRead }">{{ item.title }}</span>

<script setup lang="ts">
import { h, ref, computed } from 'vue'
import { message, Empty } from 'ant-design-vue'
import { useQCDemoStore } from '@/stores/qcDemo'
import type { QCNotification } from '@/types/quality-control'
import {
  BellOutlined,
  ReloadOutlined,
  MoreOutlined,
  EyeOutlined,
  CheckOutlined,
  FileTextOutlined,
  EyeInvisibleOutlined,
  DeleteOutlined,
  UserOutlined,
  ClockCircleOutlined,
  CloseCircleOutlined,
  WarningOutlined,
  InfoCircleOutlined,
  CheckCircleOutlined
} from '@ant-design/icons-vue'
import dayjs from 'dayjs'

const demoStore = useQCDemoStore()

// 筛选器
const filters = ref({
  type: '',
  severity: '',
  status: ''
})

// 选中的通知ID
const selectedIds = ref<string[]>([])

// 计算属性
const filteredNotifications = computed(() => {
  return demoStore.getFilteredNotifications(filters.value)
})

// 分页配置
const pagination = {
  pageSize: 10,
  showSizeChanger: true,
  showTotal: (total: number) => `共 ${total} 条`
}

// 工具函数
const getTypeColor = (type: string) => {
  const colors = {
    error: '#ff4d4f',
    warning: '#faad14',
    info: '#1890ff',
    success: '#52c41a'
  }
  return colors[type as keyof typeof colors] || '#1890ff'
}

const getTypeIcon = (type: string) => {
  const icons = {
    error: CloseCircleOutlined,
    warning: WarningOutlined,
    info: InfoCircleOutlined,
    success: CheckCircleOutlined
  }
  return icons[type as keyof typeof icons] || InfoCircleOutlined
}

const getSeverityColor = (severity: string) => {
  const colors = {
    critical: 'red',
    high: 'orange',
    medium: 'gold',
    low: 'blue'
  }
  return colors[severity as keyof typeof colors] || 'default'
}

const getSeverityLabel = (severity: string) => {
  const labels = {
    critical: '严重',
    high: '较重',
    medium: '中等',
    low: '轻微'
  }
  return labels[severity as keyof typeof labels] || severity
}

const formatTime = (timestamp: Date | string) => {
  return dayjs(timestamp).format('YYYY-MM-DD HH:mm:ss')
}

// 事件处理
const handleSelectChange = (e: any, id: string) => {
  if (e.target.checked) {
    selectedIds.value.push(id)
  } else {
    selectedIds.value = selectedIds.value.filter(selectedId => selectedId !== id)
  }
}

const handleBatchMarkRead = () => {
  demoStore.batchMarkRead(selectedIds.value)
  selectedIds.value = []
}

const handleBatchIgnore = () => {
  demoStore.batchIgnore(selectedIds.value)
  selectedIds.value = []
}

const handleBatchDelete = () => {
  demoStore.batchDelete(selectedIds.value)
  selectedIds.value = []
}

const handleView = (notification: QCNotification) => {
  demoStore.handleNotificationAction('view', notification)
}

const handleAction = (key: string, notification: QCNotification) => {
  demoStore.handleNotificationAction(key, notification)
}

const handleRefresh = () => {
  message.success('列表已刷新')
}

const handleSimulate = () => {
  demoStore.handleNotificationAction('simulate')
}
</script>
```

#### 关键点说明

1. **筛选功能**：
   - 使用 `filters` ref 存储筛选条件
   - 通过 `getFilteredNotifications` 计算属性获取筛选后的通知

2. **批量操作**：
   - 使用 `selectedIds` ref 存储选中的通知 ID
   - 批量操作后清空选中状态

3. **列表项样式**：
   - 未读通知使用蓝色背景（`unread` class）
   - 鼠标悬停时显示灰色背景

4. **空状态**：
   - 当筛选结果为空时显示空状态

```typescript
// 通知批量操作
function batchMarkRead(ids: string[]): void {
  ids.forEach(id => {
    const notification = notifications.value.find(n => n.id === id)
    if (notification) {
      notification.isRead = true
    }
  })
  message.success(`已标记 ${ids.length} 条通知为已读`)
}

function batchIgnore(ids: string[]): void {
  ids.forEach(id => {
    const notification = notifications.value.find(n => n.id === id)
    if (notification) {
      notification.status = 'ignored'
      notification.isRead = true
    }
  })
  message.success(`已忽略 ${ids.length} 条通知`)
}

function batchDelete(ids: string[]): void {
  notifications.value = notifications.value.filter(n => !ids.includes(n.id))
  message.success(`已删除 ${ids.length} 条通知`)
}

// 通知筛选
function getFilteredNotifications(filters: NotificationFilters): QCNotification[] {
  return notifications.value.filter(n => {
    if (filters.type && n.type !== filters.type) return false
    if (filters.severity && n.severity !== filters.severity) return false
    if (filters.status && n.status !== filters.status) return false
    return true
  })
}

// 申诉操作
function cancelAppeal(appealId: string): void {
  const appeal = appeals.value.find(a => a.id === appealId)
  if (appeal && appeal.status === 'pending') {
    appeal.status = 'cancelled'
    message.success('申诉已撤销')
  } else {
    message.warning('只能撤销待审核的申诉')
  }
}

function deleteAppeal(appealId: string): void {
  appeals.value = appeals.value.filter(a => a.id !== appealId)
  message.success('申诉已删除')
}

// 申诉筛选
function getFilteredAppeals(status: AppealStatus): Appeal[] {
  return appeals.value.filter(a => a.status === status)
}

// 规则参数管理
function updateRuleParams(ruleId: string, params: RuleParams): void {
  ruleParams.value.set(ruleId, params)
  message.success('规则参数已更新')
}

function getRuleParams(ruleId: string): RuleParams | undefined {
  return ruleParams.value.get(ruleId)
}

function resetRuleParams(ruleId: string): void {
  ruleParams.value.delete(ruleId)
```

## 7. 最佳实践

### 7.1 组件设计原则

1. **单一职责**：每个组件只负责一个功能模块
2. **Props 向下，Events 向上**：父子组件通信遵循单向数据流
3. **优先使用 Composition API**：使用 `<script setup>` 语法
4. **类型安全**：所有 Props、Events、Ref 都要有类型定义

### 7.2 性能优化

1. **使用 `v-show` 而非 `v-if`**（频繁切换的元素）
2. **使用 `computed` 缓存计算结果**
3. **大列表使用虚拟滚动**（Ant Design Table 自带）
4. **避免在模板中使用复杂表达式**

### 7.3 代码规范

1. **命名规范**：
   - 组件名：PascalCase
   - 变量名：camelCase
   - 常量名：UPPER_SNAKE_CASE
   - CSS 类名：kebab-case

2. **注释规范**：
   - 组件顶部添加功能说明
   - 复杂逻辑添加行内注释
   - 公共函数添加 JSDoc 注释

3. **导入顺序**：
   ```typescript
   // 1. Vue 核心
   import { ref, computed } from 'vue'
   
   // 2. 第三方库
   import { message } from 'ant-design-vue'
   
   // 3. 项目内部
   import { useQCDemoStore } from '@/stores/qcDemo'
   import type { QCNotification } from '@/types/quality-control'
   
   // 4. 图标
   import { BellOutlined } from '@ant-design/icons-vue'
   ```

```vue
<script setup lang="ts">
import { ref, watch } from 'vue'
import { message } from 'ant-design-vue'
import { useQCDemoStore } from '@/stores/qcDemo'
import type { QualityControlRule, RuleParams } from '@/types/quality-control'
import { ReloadOutlined, SaveOutlined } from '@ant-design/icons-vue'

interface Props {
  visible: boolean
  rule: QualityControlRule | null
}

interface Emits {
  (e: 'update:visible', value: boolean): void
  (e: 'save', params: RuleParams): void
}

const props = defineProps<Props>()
const emit = defineEmits<Emits>()

const demoStore = useQCDemoStore()

// 参数表单
const params = ref<RuleParams>({
  enabled: true,
  weight: 10,
  threshold: 0,
  severity: 'medium'
})

// 监听规则变化，加载参数
watch(() => props.rule, (newRule) => {
  if (newRule) {
    const savedParams = demoStore.getRuleParams(newRule.id)
    if (savedParams) {
      params.value = { ...savedParams }
    } else {
      // 使用规则的默认值
      params.value = {
        enabled: newRule.isActive ?? true,
        weight: newRule.weight ?? 10,
        threshold: newRule.threshold ?? 0,
        severity: newRule.severity ?? 'medium'
      }
    }
  }
}, { immediate: true })

// 工具函数
const getSeverityColor = (severity: string) => {
  const colors = {
    low: 'blue',
    medium: 'gold',
    high: 'orange',
    critical: 'red'
  }
  return colors[severity as keyof typeof colors] || 'default'
}

const getSeverityLabel = (severity: string) => {
  const labels = {
    low: '轻微',
    medium: '中等',
    high: '较重',
    critical: '严重'
  }
  return labels[severity as keyof typeof labels] || severity
}

// 事件处理
const handleSave = () => {
  if (props.rule) {
    demoStore.updateRuleParams(props.rule.id, params.value)
    emit('save', params.value)
    emit('update:visible', false)
  }
}

const handleReset = () => {
  if (props.rule) {
    demoStore.resetRuleParams(props.rule.id)
    // 重置为规则的默认值
    params.value = {
      enabled: props.rule.isActive ?? true,
      weight: props.rule.weight ?? 10,
      threshold: props.rule.threshold ?? 0,
      severity: props.rule.severity ?? 'medium'
    }
    message.success('已重置为默认值')
  }
}

const handleClose = () => {
  emit('update:visible', false)
}
</script>

  color: #8c8c8c;
  font-size: 12px;
  margin-top: 4px;
}
</style>
```
