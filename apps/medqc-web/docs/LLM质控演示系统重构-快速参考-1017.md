## 🔧 Store 方法速查

### 新增 Actions（通知管理）

```typescript
// 批量标记已读
demoStore.batchMarkRead(['id1', 'id2'])

// 批量忽略
demoStore.batchIgnore(['id1', 'id2'])

// 批量删除
demoStore.batchDelete(['id1', 'id2'])

// 筛选通知
const filtered = demoStore.getFilteredNotifications({
  type: 'error',
  severity: 'critical',
  status: 'pending'
})
```

### 新增 Actions（申诉管理）

```typescript
// 撤销申诉
demoStore.cancelAppeal('appeal-id')

// 删除申诉
demoStore.deleteAppeal('appeal-id')

// 筛选申诉
const filtered = demoStore.getFilteredAppeals('pending')
```

### 新增 Actions（规则参数）

```typescript
// 更新规则参数
demoStore.updateRuleParams('rule-id', {
  enabled: true,
  weight: 50,
  threshold: 80,
  severity: 'high'
})

// 获取规则参数
const params = demoStore.getRuleParams('rule-id')

// 重置规则参数
demoStore.resetRuleParams('rule-id')
```

// 格式化时间
const formatTime = (timestamp: Date | string) => {
  return dayjs(timestamp).format('YYYY-MM-DD HH:mm:ss')
}

// 相对时间
const relativeTime = (timestamp: Date | string) => {

### 工具函数

```typescript
// 获取类型颜色
const getTypeColor = (type: string) => {
  const colors = {
    error: '#ff4d4f',
    warning: '#faad14',
    info: '#1890ff',
    success: '#52c41a'
  }
  return colors[type as keyof typeof colors] || '#1890ff'
}

// 获取严重程度颜色
const getSeverityColor = (severity: string) => {
  const colors = {
    critical: 'red',
    high: 'orange',
    medium: 'gold',
    low: 'blue'
  }
  return colors[severity as keyof typeof colors] || 'default'
}

// 获取状态徽标
const getStatusBadge = (status: string) => {
  const badges = {
    pending: 'default',
    processing: 'processing',
    success: 'success',
    error: 'error'
  }
  return badges[status as keyof typeof badges] || 'default'
}
```
