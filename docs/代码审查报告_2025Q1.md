### P0-4: API 响应格式不统一

**问题描述**:
不同 API 接口返回的响应格式不一致，缺少统一的响应包装。

**所在文件**:
- `apps/medqc-web/backend/app/routers/medical_documents.py`
- `apps/medqc-web/backend/app/routers/records.py`
- `apps/medqc-web/backend/app/routers/medical_records_alias.py`

**违反规范**:
- TDD 第 4.3 节：API 接口设计 - 统一响应格式
- 开发计划第一阶段任务 T1.2.1, T1.2.2

**示例问题**:
```python
# ❌ 不一致的响应格式

# 格式 1: 直接返回字典
return {"categories": result}

# 格式 2: 包含 success 字段
return {"success": True, "document": data}

# 格式 3: 只返回 message
return {"message": "操作成功"}

# 格式 4: 包含 data 字段
return {"data": records, "total": total}
```

**修复方案**:
```python
# 1. 定义统一响应模型
class ApiResponse(BaseModel):
    success: bool
    data: Optional[Any] = None
    error: Optional[ErrorDetail] = None
    message: Optional[str] = None
    trace_id: Optional[str] = None

class ErrorDetail(BaseModel):
    code: str
    message: str
    details: Optional[List[Any]] = None

# 2. 创建响应包装器
def success_response(data: Any = None, message: str = None) -> ApiResponse:
    return ApiResponse(success=True, data=data, message=message)

def error_response(code: str, message: str, details: Any = None) -> ApiResponse:
    return ApiResponse(
        success=False,
        error=ErrorDetail(code=code, message=message, details=details)
    )

# 3. 统一所有接口
@router.get("/document-types")
async def get_document_types():
    try:
        result = await fetch_document_types()
        return success_response(data=result, message="获取成功")
    except Exception as e:
        return error_response("FETCH_ERROR", str(e))
```

**预估工作量**: 20 小时

**优先级**: P0 - 第一阶段必须完成

### P0-5: 缺少错误处理中间件

**问题描述**:
后端缺少统一的错误处理中间件，异常处理不规范。

**所在文件**:
- `apps/medqc-web/backend/app/main.py`

**违反规范**:
- TDD 第 7.3 节：API 安全 - 请求验证
- TDD 第 9.1 节：代码质量 - 错误处理

**影响**:
- 错误信息泄露：可能暴露敏感的堆栈信息
- 用户体验差：错误信息不友好
- 难以调试：缺少统一的错误日志

**修复方案**:
```python
# 1. 创建统一错误处理中间件
from fastapi import Request, status
from fastapi.responses import JSONResponse
from fastapi.exceptions import RequestValidationError
from starlette.exceptions import HTTPException as StarletteHTTPException

@app.exception_handler(StarletteHTTPException)
async def http_exception_handler(request: Request, exc: StarletteHTTPException):
    return JSONResponse(
        status_code=exc.status_code,
        content={
            "success": False,
            "error": {
                "code": f"HTTP_{exc.status_code}",
                "message": exc.detail
            },
            "trace_id": request.state.trace_id
        }
    )

@app.exception_handler(RequestValidationError)
async def validation_exception_handler(request: Request, exc: RequestValidationError):
    return JSONResponse(
        status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
        content={
            "success": False,
            "error": {
                "code": "VALIDATION_ERROR",
                "message": "请求参数验证失败",
                "details": exc.errors()
            }
        }
    )

@app.exception_handler(Exception)
async def general_exception_handler(request: Request, exc: Exception):
    logger.error(f"Unhandled exception: {exc}", exc_info=True)
    return JSONResponse(
        status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
        content={
            "success": False,
            "error": {
                "code": "INTERNAL_ERROR",
                "message": "服务器内部错误"
            }
        }
    )
```

**预估工作量**: 8 小时

**优先级**: P0 - 第一阶段必须完成

### P1-3: 缺少 API 请求/响应日志

**问题描述**:
后端缺少统一的请求/响应日志记录，难以调试和监控。

**违反规范**:
- TDD 第 9.4 节：监控与可观测性

**修复方案**:
```python
# 添加请求日志中间件
@app.middleware("http")
async def log_requests(request: Request, call_next):
    trace_id = str(uuid.uuid4())
    request.state.trace_id = trace_id
    
    start_time = time.time()
    logger.info(f"[{trace_id}] {request.method} {request.url.path}")
    
    response = await call_next(request)
    
    duration = time.time() - start_time
    logger.info(
        f"[{trace_id}] {request.method} {request.url.path} "
        f"- {response.status_code} - {duration:.3f}s"
    )
    
    response.headers["X-Trace-ID"] = trace_id
    return response
```

**预估工作量**: 4 小时

```typescript
// src/services/llm/BaseLLMClient.ts
export abstract class BaseLLMClient {
  protected config: LLMConfig

  constructor(config: LLMConfig) {
    this.config = config
  }

  abstract initialize(): Promise<boolean>
  abstract callAPI(prompt: string, options?: CallOptions): Promise<LLMResponse>
  abstract extractMedicalData(text: string): Promise<ExtractedMedicalData>

  // 通用方法
  protected validateConfig(): boolean {
    return !!(this.config.apiKey && this.config.baseURL)
  }

  protected handleError(error: any): never {
    logger.error('LLM API Error:', error)
    throw new LLMError(error.message)
  }
}

// src/services/llm/OpenAIClient.ts
export class OpenAIClient extends BaseLLMClient {
  async initialize(): Promise<boolean> {
    if (!this.validateConfig()) {
      return false
    }
    // OpenAI 特定的初始化逻辑
    return true
  }

  async callAPI(prompt: string, options?: CallOptions): Promise<LLMResponse> {
    try {
      const response = await fetch(`${this.config.baseURL}/chat/completions`, {
        method: 'POST',
        headers: {
          'Authorization': `Bearer ${this.config.apiKey}`,
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({
          model: this.config.model,
          messages: [{ role: 'user', content: prompt }],
          ...options
        })
      })
      return await response.json()
    } catch (error) {
      this.handleError(error)
    }
  }

  async extractMedicalData(text: string): Promise<ExtractedMedicalData> {
    const prompt = this.buildExtractionPrompt(text)
    const response = await this.callAPI(prompt)
    return this.parseResponse(response)
  }
}

// src/services/llm/index.ts
export class LLMServiceFacade {
  private client: BaseLLMClient

  async initialize(provider: 'openai' | 'deepseek' | 'aliyun', config: LLMConfig) {
    switch (provider) {
      case 'openai':
        this.client = new OpenAIClient(config)
        break
      case 'deepseek':
        this.client = new DeepSeekClient(config)
        break
      case 'aliyun':
        this.client = new AliyunClient(config)
        break
    }
    return await this.client.initialize()
  }

  async extractMedicalData(text: string): Promise<ExtractedMedicalData> {
    if (!this.client) {
      throw new Error('LLM Service not initialized')
    }
    return await this.client.extractMedicalData(text)
  }
}

export const llmService = new LLMServiceFacade()
```

### B.2 统一 API 响应格式

```python
# app/core/response.py
from typing import Any, Optional, List
from pydantic import BaseModel
from fastapi.responses import JSONResponse

class ErrorDetail(BaseModel):
    """错误详情"""
    code: str
    message: str
    details: Optional[List[Any]] = None

class ApiResponse(BaseModel):
    """统一API响应格式"""
    success: bool
    data: Optional[Any] = None
    error: Optional[ErrorDetail] = None
    message: Optional[str] = None
    trace_id: Optional[str] = None

def success_response(
    data: Any = None,
    message: Optional[str] = None,
    trace_id: Optional[str] = None
) -> JSONResponse:
    """成功响应"""
    response = ApiResponse(
        success=True,
        data=data,
        message=message,
        trace_id=trace_id
    )
    return JSONResponse(content=response.dict())

def error_response(
    code: str,
    message: str,
    details: Any = None,
    status_code: int = 400,
    trace_id: Optional[str] = None
) -> JSONResponse:
    """错误响应"""
    response = ApiResponse(
        success=False,
        error=ErrorDetail(code=code, message=message, details=details),
        trace_id=trace_id
    )
    return JSONResponse(content=response.dict(), status_code=status_code)

# 使用示例
# app/routers/medical_documents.py
from app.core.response import success_response, error_response

@router.get("/document-types")
async def get_document_types(request: Request):
    try:
        result = await fetch_document_types()
        return success_response(
            data=result,
            message="获取文档类型成功",
            trace_id=request.state.trace_id
        )
    except ValueError as e:
        return error_response(
            code="VALIDATION_ERROR",
            message=str(e),
            status_code=400,
            trace_id=request.state.trace_id
        )
    except Exception as e:
        logger.error(f"获取文档类型失败: {e}", exc_info=True)
        return error_response(
            code="INTERNAL_ERROR",
            message="服务器内部错误",
            status_code=500,
            trace_id=request.state.trace_id
        )
```
